# About Me

<div class="compact-section" markdown="1">

I am a Ph.D. student in Computer Science at [Emory University](https://catalog.college.emory.edu/academics/departments/computer-science.html), advised by [Prof. Liang Zhao](https://cs.emory.edu/~lzhao41/). My research focuses on learning theory, statistical machine learning, optimization, efficient model design and training, semi-supervised learning, and AI agents.

Before Emory, I pursued Ph.D. studies in Computer Science at the [University of Texas at Arlington](https://www.uta.edu/academics/schools-colleges/engineering/academics/departments/cse), advised by [Prof. Yingying Zhu](https://zyy123jy.github.io/My-Web-Sites/). At the [National University of Singapore](https://www.stat.nus.edu.sg/), I pursued Ph.D. studies and received an M.S. in Statistics under the supervision of [Prof. Cheng Li](https://www.stat.nus.edu.sg/faculty_member/li-cheng/). I earned B.S. degrees in Physics and Mathematics from [Peking University](https://english.pku.edu.cn/).

### Research Interests

<ul class="research-interests">
  <li><strong>Theory, Statistical Machine Learning, and Optimization</strong> &mdash; generalization, nonconvex landscapes, and principled learning algorithms</li>
  <li><strong>Model Optimization</strong> &mdash; neural networks, LLM architecture design, pre-training, and post-training</li>
  <li><strong>Semi-Supervised Learning and AI Agents</strong> &mdash; learning with limited labels and building reliable agents with world models and memory</li>
</ul>

My long-term goal is to develop principled and practical learning systems that remain efficient, reliable, and understandable under real-world constraints.

<script>
(() => {
  const list = document.getElementById("visitor-log-list");
  const pins = document.getElementById("visitor-map-pins");
  if (!list || !pins) return;
  const relativeTime = (seconds) => {
    const value = Math.max(0, Number(seconds) || 0);
    if (value < 60) return "just now";
    if (value < 3600) return `${Math.floor(value / 60)} min ago`;
    if (value < 86400) return `${Math.floor(value / 3600)} hr ago`;
    return `${Math.floor(value / 86400)} day${value < 172800 ? "" : "s"} ago`;
  };
  const countryFlag = (code) => {
    if (!code || code.length !== 2) return "🌐";
    return String.fromCodePoint(...code.toUpperCase().split("").map((letter) => 127397 + letter.charCodeAt(0)));
  };
  const placeLabel = (visitor) => [visitor.city, visitor.region, visitor.country]
    .filter((value, position, values) => value && values.indexOf(value) === position)
    .join(", ") || "Unknown region";
  const secondsSince = (visitor, updatedAt) => {
    const timestamp = Date.parse(visitor.last_seen || updatedAt || "");
    return Number.isFinite(timestamp) ? Math.max(0, (Date.now() - timestamp) / 1000) : visitor.age_s;
  };
  const render = (data) => {
    const visitors = Array.isArray(data.visitors) ? [...data.visitors] : [];
    visitors.sort((left, right) => {
      const liveDifference = Number(right.status === "live") - Number(left.status === "live");
      if (liveDifference) return liveDifference;
      return Date.parse(right.last_seen || data.updated_at || "") - Date.parse(left.last_seen || data.updated_at || "");
    });
    list.replaceChildren();
    pins.replaceChildren();
    if (!visitors.length) {
      const empty = document.createElement("li");
      empty.className = "visitor-log-empty";
      empty.textContent = "No visits yet.";
      list.appendChild(empty);
      return;
    }
    const plottedLocations = new Set();
    visitors.forEach((visitor, index) => {
      const latitude = Number(visitor.lat);
      const longitude = Number(visitor.lon);
      if (Number.isFinite(latitude) && Number.isFinite(longitude)) {
        const locationKey = `${latitude.toFixed(3)},${longitude.toFixed(3)}`;
        if (!plottedLocations.has(locationKey)) {
          plottedLocations.add(locationKey);
          const pin = document.createElement("i");
          pin.className = visitor.status === "live" ? "visitor-map-pin is-live" : "visitor-map-pin";
          pin.style.left = `${Math.min(100, Math.max(0, ((longitude + 180) / 360) * 100))}%`;
          pin.style.top = `${Math.min(100, Math.max(0, ((90 - latitude) / 180) * 100))}%`;
          pin.title = placeLabel(visitor);
          pins.appendChild(pin);
        }
      }

      const item = document.createElement("li");
      const identity = document.createElement("span");
      const location = document.createElement("span");
      const time = document.createElement("time");
      identity.className = "visitor-log-identity";
      location.className = "visitor-log-location";
      time.className = "visitor-log-time";
      identity.textContent = `Visitor ${String(index + 1).padStart(2, "0")}`;
      location.textContent = `${countryFlag(visitor.country_code)} ${placeLabel(visitor)}`;
      time.dateTime = visitor.last_seen || "";
      time.textContent = Number(visitor.live_visits) > 0 ? "Live" : relativeTime(secondsSince(visitor, data.updated_at));
      item.append(identity, location, time);
      list.appendChild(item);
    });
  };
  const visitorApi = "https://xinyuan-visitor-api.opendelta4.chatgpt.site/api/visitors";
  const locateVisitor = async () => {
    const response = await fetch("https://ipwho.is/", { cache: "no-store" });
    if (!response.ok) throw new Error("Location unavailable");
    const location = await response.json();
    if (location.success === false) throw new Error("Location unavailable");
    return {
      city: location.city,
      region: location.region,
      country: location.country,
      country_code: location.country_code,
      latitude: location.latitude,
      longitude: location.longitude
    };
  };
  let refreshing = false;
  let lastRecordedAt = 0;
  const refresh = async (recordVisit = false) => {
    if (refreshing) return;
    refreshing = true;
    try {
      const request = { cache: "no-store" };
      if (recordVisit) {
        request.method = "POST";
        request.headers = { "content-type": "application/json" };
        request.body = JSON.stringify(await locateVisitor().catch(() => ({})));
      }
      const response = await fetch(`${visitorApi}?v=${Date.now()}`, request);
      if (!response.ok) throw new Error("Visitor data unavailable");
      render(await response.json());
      if (recordVisit) lastRecordedAt = Date.now();
    } catch (_error) {
      const empty = document.createElement("li");
      empty.className = "visitor-log-empty";
      empty.textContent = "Visitor list is temporarily unavailable.";
      list.replaceChildren(empty);
    } finally {
      refreshing = false;
    }
  };
  refresh(true);
  window.setInterval(() => refresh(false), 60 * 1000);
  document.addEventListener("visibilitychange", () => {
    if (!document.hidden) refresh(Date.now() - lastRecordedAt >= 10 * 60 * 1000);
  });
})();
</script>
